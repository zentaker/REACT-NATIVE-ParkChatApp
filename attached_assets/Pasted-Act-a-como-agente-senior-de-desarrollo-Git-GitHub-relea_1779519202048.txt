Quinta tarea: preparar release automation.

Revisar si existe:

.github/workflows/release.yml

Si no existe, crearlo.

Contenido esperado:

name: Release

on:
  push:
    tags:
      - "v*"

permissions:
  contents: write

jobs:
  release:
    name: Create GitHub Release
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Create source bundle
        run: |
          mkdir -p dist
          zip -r dist/shadowlia_source_bundle.zip . \
            -x ".git/*" \
            -x "dist/*" \
            -x "models/*" \
            -x "outputs/*" \
            -x "*.wav" \
            -x "*.mp3" \
            -x "*.flac" \
            -x ".env" \
            -x ".env.*" \
            -x "__pycache__/*" \
            -x ".venv/*" \
            -x ".cache/*" \
            -x ".pytest_cache/*" \
            -x ".ipynb_checkpoints/*"

      - name: Create release
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh release create "${GITHUB_REF_NAME}" \
            dist/shadowlia_source_bundle.zip \
            --title "Shadowlia ${GITHUB_REF_NAME}" \
            --generate-notes
